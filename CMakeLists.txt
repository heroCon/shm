cmake_minimum_required(VERSION 3.10)
project(shm_pubsub VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
find_package(Threads REQUIRED)

add_library(shm_pubsub INTERFACE)
add_library(shm_pubsub::shm_pubsub ALIAS shm_pubsub)
target_include_directories(shm_pubsub INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(shm_pubsub INTERFACE atomic Threads::Threads rt)

foreach(program publisher subscriber)
  add_executable(${program} ${program}.cpp)
  target_link_libraries(${program} PRIVATE shm_pubsub)
endforeach()
add_executable(broadcast_example examples/broadcast.cpp)
target_link_libraries(broadcast_example PRIVATE shm_pubsub)
add_executable(competing_example examples/competing.cpp)
target_link_libraries(competing_example PRIVATE shm_pubsub)

include(CTest)
if(BUILD_TESTING)
  foreach(test free_list_test atomic_queue_test semantics_test multiprocess_test)
    add_executable(${test} tests/${test}.cpp)
    target_link_libraries(${test} PRIVATE shm_pubsub)
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 20)
  endforeach()
endif()
option(SHM_PUBSUB_BUILD_BENCHMARKS "Build reproducible benchmarks" OFF)
if(SHM_PUBSUB_BUILD_BENCHMARKS)
  add_executable(shm_benchmark benchmarks/shm_benchmark.cpp)
  add_executable(uds_benchmark benchmarks/uds_benchmark.cpp)
  target_link_libraries(shm_benchmark PRIVATE shm_pubsub)
  target_link_libraries(uds_benchmark PRIVATE Threads::Threads)
endif()
install(FILES shm_pubsub.h lock_free_list.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS shm_pubsub EXPORT shm_pubsubTargets)
install(EXPORT shm_pubsubTargets NAMESPACE shm_pubsub:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/shm_pubsub)
